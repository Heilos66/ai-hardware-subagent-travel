cmake_minimum_required(VERSION 3.16)
project(travel_ame_bench CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT AME_SOURCE)
  message(FATAL_ERROR "Set -DAME_SOURCE=/path/to/ame")
endif()
add_subdirectory("${AME_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_executable(ame_kb_worker worker.cpp)
target_link_libraries(ame_kb_worker PRIVATE ame_engine)
if(TARGET ame_mem)
  target_link_libraries(ame_kb_worker PRIVATE ame_mem)
  target_compile_definitions(ame_kb_worker PRIVATE AME_HAS_LMEM=1)
endif()
if(EXISTS "${AME_SOURCE}/android/app/src/main/assets/cheap_model.json")
  target_compile_definitions(ame_kb_worker PRIVATE AME_HAS_CHEAP=1)
endif()
